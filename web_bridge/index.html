How to integrate this into a Custom Chrome Extension
You can forward tab updates, notification badges, or media states to Presto from your own Chrome Extension background service worker: 
// In your Chrome extension service worker or offscreen document:
// 1. Listen to tab updates (e.g., Jira, GitHub, Slack)
chrome.tabs.onUpdated.addListener((tabId, changeInfo, tab) => {
  if (tab.title && tab.title.includes("Pull Request")) {
    bridgePort.postMessage({ type: "notify", text: "New PR Activity" });
  }
});

// 2. Listen to Presto button presses received over Web Bluetooth:
function onPrestoButtonPress(event) {
  if (event.action === "mute_tab") {
    chrome.tabs.query({ active: true, currentWindow: true }, (tabs) => {
      chrome.tabs.update(tabs[0].id, { muted: true });
    });
  } else if (event.event === "notification_tap") {
    console.log("User tapped notification on Presto!");
  }
}
